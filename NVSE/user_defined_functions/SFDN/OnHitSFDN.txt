    let iTestCount += 1
    PrintV iTestCount
end

int iDrawX
int iDrawY
float fScreenResolutionWidth
float fScreenResolutionHeight
begin Function ;Hit coordinates to screen coordinates
	let array_var aHit3DData := rTarget.GetHit3DData 0
	let float fHitX := aHit3DData[0]
	let float fHitY := aHit3DData[1]
	let float fHitZ := aHit3DData[2]

	int xOut
	int yOut
	int zOut
	let int isOnScreen := WorldToScreen (xOut) (yOut) (zOut) fHitX fHitY fHitZ 0 

	let fScreenResolutionWidth := GetScreenTrait 0
	let fScreenResolutionHeight := GetScreenTrait 1
	let iDrawX := fScreenResolutionWidth * xOut
	let iDrawY := fScreenResolutionHeight * yOut
end

    let float fDamageNumber := rTarget.GetHitHealthDamage

	InjectUIComponent "HUDMainMenu" "%Z" sFloatingNumberXML
	let sUIElement := Sv_Construct "HUDMainMenu\damageNumberContainer_%g" iUID
	let sUIElementPath := Sv_Construct "HUDMainMenu\damageNumberContainer_%g\damageNumberText" iUID
end

Begin Function ; Animation
    let float fStartX := iDrawX
    let float fStartY := iDrawY

    let float fTime := 0
    let float fHeight := 50
    
    let int iRandomDirection := GetRandomInRange 0 2
	if iRandomDirection
		let float fWidth := 50
	else
		let float fWidth := -50
	endif

	let ref	rUpdatePositionLambda := (begin Function {}
		int xOut
		int yOut
		int zOut
		let int isOnScreen := WorldToScreen (xOut) (yOut) (zOut) fHitX fHitY fHitZ 0 

		let fStartX := fScreenResolutionWidth * xOut
		let fStartY := fScreenResolutionHeight * yOut
	end)

    let ref rAnimationLambda := (begin Function { }
        let fTime += GetSecondsPassed
        let float fT := fTime / fAnimationDuration

        let iDrawX := fStartX + (fWidth * fT)
        let iDrawY := fStartY - (fHeight * 4 * fT * (1 - fT))

		SetUIFloatAlt ((Sv_Construct "%Z\x" sUIElementPath)) iDrawX
		SetUIFloatAlt ((Sv_Construct "%Z\y" sUIElementPath)) iDrawY
    end)

	CallForSeconds fAnimationDuration rUpdatePositionLambda 1
    CallForSeconds fAnimationDuration rAnimationLambda 1
End

Begin Function ; Animate floating number
	SetUIFLoatGradual ((Sv_Construct "%Z\zoom" sUIElementPath)) 100 20 1 
	SetUIFLoatGradual ((Sv_Construct "%Z\alpha" sUIElementPath)) 255 0 1
end

Begin Function ;Clean up
	CallAfterSeconds fAnimationDuration ({} => UnloadUIComponent (sUIElement))
end
